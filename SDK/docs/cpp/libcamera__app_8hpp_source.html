133 class MessageQueue
134 {
135 public:
136 template <typename U>
137 void Post(U &&msg)
138 {
139 std::unique_lock<std::mutex> lock(mutex_);
140 queue_.push(std::forward<U>(msg));
141 cond_.notify_one();
142 }
143 T Wait()
144 {
145 std::unique_lock<std::mutex> lock(mutex_);
146 cond_.wait(lock, [this] { return !queue_.empty(); });
147 T msg = std::move(queue_.front());
148 queue_.pop();
149 return msg;
150 }

153 std::unique_lock<std::mutex> lock(mutex_);
154 queue_ = {};
155 }

192struct FrameInfo
193{
194 FrameInfo(libcamera::ControlList &ctrls)
195 : exposure_time(0.0), digital_gain(0.0), colour_gains({ { 0.0f, 0.0f } }), focus(0.0), aelock(false)
196 {
197 auto exp = ctrls.get(libcamera::controls::ExposureTime);
198 if (exp)
199 exposure_time = *exp;
200
201 auto ag = ctrls.get(libcamera::controls::AnalogueGain);
202 if (ag)
203 analogue_gain = *ag;
204
205 auto dg = ctrls.get(libcamera::controls::DigitalGain);
206 if (dg)
207 digital_gain = *dg;
208
209 auto cg = ctrls.get(libcamera::controls::ColourGains);
210 if (cg)
211 {
212 colour_gains[0] = (*cg)[0], colour_gains[1] = (*cg)[1];
213 }
214
215 auto fom = ctrls.get(libcamera::controls::FocusFoM);
216 if (fom)
217 focus = *fom;
218
219 auto ae = ctrls.get(libcamera::controls::AeLocked);
220 if (ae)
221 aelock = *ae;
222 }
223
224 std::string ToString(std::string &info_string) const
225 {
226 std::string parsed(info_string);
227
228 for (auto const &t : tokens)
229 {
230 std::size_t pos = parsed.find(t);
231 if (pos != std::string::npos)
232 {
233 std::stringstream value;
234 value << std::fixed << std::setprecision(2);
235
236 if (t == "%frame")
237 value << sequence;
238 else if (t == "%fps")
239 value << fps;
240 else if (t == "%exp")
241 value << exposure_time;
242 else if (t == "%ag")
243 value << analogue_gain;
244 else if (t == "%dg")
245 value << digital_gain;
246 else if (t == "%rg")
247 value << colour_gains[0];
248 else if (t == "%bg")
249 value << colour_gains[1];
250 else if (t == "%focus")
251 value << focus;
252 else if (t == "%aelock")
253 value << aelock;
254
255 parsed.replace(pos, t.length(), value.str());
256 }
257 }
258
259 return parsed;
260 }
261
262 unsigned int sequence;
263 float exposure_time;
264 float analogue_gain;
265 float digital_gain;
266 std::array<float, 2> colour_gains;
267 float focus;
268 float fps;
269 bool aelock;
270
271private:
272 // Info text tokens.
273 inline static const std::string tokens[] = { "%frame", "%fps", "%exp", "%ag", "%dg",
274 "%rg", "%bg", "%focus", "%aelock" };
275};

282 Metadata(Metadata const &other)

294
295 template <typename T>
296 void Set(std::string const &tag, T &&value)
297 {
298 std::scoped_lock lock(mutex_);
299 data_.insert_or_assign(tag, std::forward<T>(value));
300 }
301
302 template <typename T>
303 int Get(std::string const &tag, T &value) const
304 {
305 std::scoped_lock lock(mutex_);
306 auto it = data_.find(tag);
307 if (it == data_.end())
308 return -1;
309 value = std::any_cast<T>(it->second);
310 return 0;
311 }
312
313 void Clear()
314 {
315 std::scoped_lock lock(mutex_);
316 data_.clear();
317 }
318
319 Metadata &operator=(Metadata const &other)
320 {
321 std::scoped_lock lock(mutex_, other.mutex_);
322 data_ = other.data_;
323 return *this;
324 }
325
326 Metadata &operator=(Metadata &&other)
327 {
328 std::scoped_lock lock(mutex_, other.mutex_);
329 data_ = std::move(other.data_);
330 other.data_.clear();
331 return *this;
332 }
333
334 void Merge(Metadata &other)
335 {
336 std::scoped_lock lock(mutex_, other.mutex_);
337 data_.merge(other.data_);
338 }
339
340 template <typename T>
341 T *GetLocked(std::string const &tag)
342 {
343 // This allows in-place access to the Metadata contents,
344 // for which you should be holding the lock.
345 auto it = data_.find(tag);
346 if (it == data_.end())
347 return nullptr;
348 return std::any_cast<T>(&it->second);
349 }
350
351 template <typename T>
352 void SetLocked(std::string const &tag, T &&value)
353 {
354 // Use this only if you're holding the lock yourself.
355 data_.insert_or_assign(tag, std::forward<T>(value));
356 }
357
358 // Note: use of (lowercase) lock and unlock means you can create scoped
359 // locks with the standard lock classes.
360 // e.g. std::lock_guard<RPiController::Metadata> lock(metadata)
361 void lock() { mutex_.lock(); }
362 void unlock() { mutex_.unlock(); }
363
364private:
365 mutable std::mutex mutex_;
366 std::map<std::string, std::any> data_;
367};
